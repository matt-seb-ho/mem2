import numpy as np

def transform(input_grid: np.ndarray) -> np.ndarray:
    """
    Transforms a 2x2 input grid into a 6x6 output grid by tiling the original
    and its horizontally flipped version in a 3x3 block pattern:
    - Top row of 2x2 blocks: original input repeated horizontally 3 times.
    - Middle row of 2x2 blocks: horizontally flipped input repeated horizontally 3 times.
    - Bottom row of 2x2 blocks: original input repeated horizontally 3 times.
    """
    # Repeat the original input horizontally 3 times -> shape (2, 6)
    top_block = np.tile(input_grid, (1, 3))
    # Horizontally flip the input (swap columns) and repeat horizontally 3 times -> shape (2, 6)
    middle_block = np.tile(np.fliplr(input_grid), (1, 3))
    # Stack the three blocks vertically: top, middle, top -> shape (6, 6)
    output_grid = np.vstack([top_block, middle_block, top_block])
    return output_grid
