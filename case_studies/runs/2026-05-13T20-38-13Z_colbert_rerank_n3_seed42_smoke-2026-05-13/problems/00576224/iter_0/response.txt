import numpy as np

def transform(input_grid: np.ndarray) -> np.ndarray:
    """
    Transform a 2x2 input grid into a 6x6 output grid by tiling the input
    in a 3x3 pattern. The middle row of tiles (block row index 1) uses the
    horizontally flipped version of the input; the top and bottom rows use
    the original input.
    """
    # Ensure input is 2x2
    assert input_grid.shape == (2, 2), "Input grid must be 2x2"
    
    # Create output grid (6x6) initialized with zeros (black)
    output = np.zeros((6, 6), dtype=int)
    
    # Precompute flipped version
    flipped = np.fliplr(input_grid)
    
    # Tile the 2x2 blocks
    for r in range(3):          # block row index
        for c in range(3):      # block column index
            # Choose tile: original for even rows, flipped for odd rows
            tile = input_grid if r % 2 == 0 else flipped
            # Place tile in the appropriate 2x2 region
            output[2*r:2*r+2, 2*c:2*c+2] = tile
    
    return output
