import numpy as np

def transform(input_grid: np.ndarray) -> np.ndarray:
    """
    Transform the input grid by recoloring all azure (8) pixels to a new color
    determined by the shape of the blue (1) plus sign, and removing the blue shape.
    """
    # Find all blue pixels (color 1)
    blue_coords = np.argwhere(input_grid == 1)
    if len(blue_coords) == 0:
        # No blue shape – just return a copy (should not happen in this puzzle)
        return input_grid.copy()
    
    # Compute bounding box of blue pixels
    min_row = blue_coords[:, 0].min()
    max_row = blue_coords[:, 0].max()
    min_col = blue_coords[:, 1].min()
    max_col = blue_coords[:, 1].max()
    
    # The blue shape is always a 3x3 pattern (as seen in examples)
    # Build a set of relative coordinates within the bounding box
    relative_coords = frozenset(
        (r - min_row, c - min_col) for r, c in blue_coords
    )
    
    # Define known patterns and their corresponding target colors
    # Pattern A: top row: left and right; middle: center; bottom: all three
    pattern_A = frozenset({(0,0), (0,2), (1,1), (2,0), (2,1), (2,2)})
    # Pattern B: top row: all three; middle: left and right; bottom: center
    pattern_B = frozenset({(0,0), (0,1), (0,2), (1,0), (1,2), (2,1)})
    # Pattern C: perfect plus (center and four arms)
    pattern_C = frozenset({(0,1), (1,0), (1,1), (1,2), (2,1)})
    
    if relative_coords == pattern_A:
        target_color = 3   # green
    elif relative_coords == pattern_B:
        target_color = 7   # orange
    elif relative_coords == pattern_C:
        target_color = 2   # red
    else:
        # Fallback: if unknown pattern, use a default (should not happen)
        target_color = 2
    
    # Create output grid as a copy of the input
    output_grid = input_grid.copy()
    
    # Recolor azure (8) to target_color, and remove blue (1) to black (0)
    output_grid[output_grid == 8] = target_color
    output_grid[output_grid == 1] = 0
    
    return output_grid
