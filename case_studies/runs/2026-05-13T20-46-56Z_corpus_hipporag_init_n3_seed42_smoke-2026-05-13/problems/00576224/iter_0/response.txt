import numpy as np

def transform(input_grid: np.ndarray) -> np.ndarray:
    """
    Transforms a 2x2 input grid into a 6x6 output grid by tiling the input
    and its horizontal flip in alternating rows of blocks.
    
    The output is a 3x3 tiling of 2x2 blocks. Blocks in rows 0 and 2 (0-indexed)
    are the original input, while blocks in row 1 are the input flipped horizontally.
    """
    h, w = input_grid.shape  # expected h=2, w=2
    out_h, out_w = h * 3, w * 3
    output = np.zeros((out_h, out_w), dtype=int)
    
    for i in range(out_h):
        for j in range(out_w):
            # Determine which block row/column we are in
            block_row = i // h
            block_col = j // w  # not actually used, but kept for clarity
            # Local coordinates within the 2x2 block
            r = i % h
            c = j % w
            
            if block_row % 2 == 0:
                # Use original input
                output[i, j] = input_grid[r, c]
            else:
                # Use horizontally flipped input (swap columns)
                output[i, j] = input_grid[r, w - 1 - c]
    
    return output
