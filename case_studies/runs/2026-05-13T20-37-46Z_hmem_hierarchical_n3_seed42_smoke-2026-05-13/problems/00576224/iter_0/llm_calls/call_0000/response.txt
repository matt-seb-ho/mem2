import numpy as np

def transform(input_grid: np.ndarray) -> np.ndarray:
    """
    Transforms a 2x2 input grid into a 6x6 output grid by tiling it in a 3x3 pattern,
    where the middle row of tiles is horizontally flipped.
    """
    h, w = input_grid.shape
    # Output dimensions are 3 times the input dimensions
    output = np.zeros((h * 3, w * 3), dtype=input_grid.dtype)
    
    for i in range(3):          # block rows
        for j in range(3):      # block columns
            # Middle row of blocks (i == 1) uses horizontally flipped input
            block = np.fliplr(input_grid) if i == 1 else input_grid
            output[i*h:(i+1)*h, j*w:(j+1)*w] = block
    
    return output
