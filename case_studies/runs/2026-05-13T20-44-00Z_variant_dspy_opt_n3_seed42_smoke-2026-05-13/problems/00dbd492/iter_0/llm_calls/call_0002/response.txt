import numpy as np

def transform(input_grid: np.ndarray) -> np.ndarray:
    """
    Fills the interior of red-bordered rectangles that contain a single red cell
    at their center. The fill color depends on the side length of the rectangle:
    - 5x5 -> 8 (light blue)
    - 7x7 -> 4 (yellow)
    - 9x9 -> 3 (green)
    """
    output_grid = np.copy(input_grid)
    rows, cols = input_grid.shape

    # mapping from side length (odd) to fill color
    fill_map = {5: 8, 7: 4, 9: 3}

    # iterate over all possible top-left corners and odd side lengths
    for r in range(rows):
        for c in range(cols):
            for L in [3, 5, 7, 9, 11, 13, 15, 17, 19]:  # odd sizes up to max dimension
                if r + L > rows or c + L > cols:
                    continue
                # check border: all cells on the perimeter must be red (2)
                border_ok = True
                for i in range(r, r + L):
                    for j in range(c, c + L):
                        if i == r or i == r + L - 1 or j == c or j == c + L - 1:
                            if input_grid[i, j] != 2:
                                border_ok = False
                                break
                    if not border_ok:
                        break
                if not border_ok:
                    continue

                # check interior: all cells strictly inside must be black (0)
                # except the center cell which must be red (2)
                center_r = r + L // 2
                center_c = c + L // 2
                interior_ok = True
                for i in range(r + 1, r + L - 1):
                    for j in range(c + 1, c + L - 1):
                        if i == center_r and j == center_c:
                            if input_grid[i, j] != 2:
                                interior_ok = False
                                break
                        else:
                            if input_grid[i, j] != 0:
                                interior_ok = False
                                break
                    if not interior_ok:
                        break
                if not interior_ok:
                    continue

                # valid rectangle found, fill interior with appropriate color
                fill_color = fill_map.get(L)
                if fill_color is None:
                    # unknown size, skip (should not happen in test)
                    continue
                for i in range(r + 1, r + L - 1):
                    for j in range(c + 1, c + L - 1):
                        if not (i == center_r and j == center_c):
                            output_grid[i, j] = fill_color

    return output_grid
