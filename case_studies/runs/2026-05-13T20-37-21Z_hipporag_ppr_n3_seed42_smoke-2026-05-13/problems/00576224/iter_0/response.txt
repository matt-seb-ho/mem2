import numpy as np

def transform(input_grid: np.ndarray) -> np.ndarray:
    # input_grid is a 2x2 numpy array
    # Create horizontal tiling of the original input (3 times)
    original_tiled = np.tile(input_grid, (1, 3))  # shape (2, 6)
    # Create the column-swapped version (reverse column order)
    swapped = input_grid[:, ::-1]
    swapped_tiled = np.tile(swapped, (1, 3))      # shape (2, 6)
    # Stack vertically: original, swapped, original
    output_grid = np.vstack([original_tiled, swapped_tiled, original_tiled])
    return output_grid
