import numpy as np

def transform(input_grid: np.ndarray) -> np.ndarray:
    """
    Transforms a 2x2 input grid into a 6x6 output grid by tiling the input
    and its horizontal reflection in a 3x3 pattern:
    - Top and bottom rows of blocks: original input
    - Middle row of blocks: horizontally flipped input
    Each block is repeated three times horizontally.
    """
    h, w = input_grid.shape
    # Create the three horizontal strips
    top = np.hstack([input_grid] * 3)
    middle = np.hstack([np.fliplr(input_grid)] * 3)
    bottom = np.hstack([input_grid] * 3)
    # Stack vertically
    output = np.vstack([top, middle, bottom])
    return output
