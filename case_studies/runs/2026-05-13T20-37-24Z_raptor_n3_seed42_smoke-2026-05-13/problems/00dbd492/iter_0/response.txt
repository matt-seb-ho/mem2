import numpy as np

def transform(input_grid: np.ndarray) -> np.ndarray:
    """
    Fill the interiors of all red (color 2) rectangular frames (border thickness 1)
    with a color determined by the interior size:
      3x3 interior  → color 8
      5x5 interior  → color 4
      7x7 interior  → color 3
    Interior black cells are filled; other colors (e.g., red interior objects) remain.
    """
    output = input_grid.copy()
    h, w = input_grid.shape
    
    # Map from interior side length (assumes square interior) to fill colour
    fill_map = {3: 8, 5: 4, 7: 3}
    
    # Detect all red rectangular frames (border of 2's, thickness 1)
    for r1 in range(h):
        for c1 in range(w):
            if input_grid[r1, c1] != 2:
                continue
            # Potential top-left corner found, try to extend to bottom-right
            for r2 in range(r1 + 2, h):   # at least 1 cell interior height
                if input_grid[r2, c1] != 2:
                    continue
                for c2 in range(c1 + 2, w):  # at least 1 cell interior width
                    if input_grid[r1, c2] != 2 or input_grid[r2, c2] != 2:
                        continue
                    
                    # Check top and bottom borders are solid red
                    top_ok = np.all(input_grid[r1, c1:c2+1] == 2)
                    bottom_ok = np.all(input_grid[r2, c1:c2+1] == 2)
                    left_ok = np.all(input_grid[r1:r2+1, c1] == 2)
                    right_ok = np.all(input_grid[r1:r2+1, c2] == 2)
                    
                    if not (top_ok and bottom_ok and left_ok and right_ok):
                        continue
                    
                    # Compute interior dimensions
                    interior_h = r2 - r1 - 1
                    interior_w = c2 - c1 - 1
                    
                    # Only handle square interiors (as per examples)
                    if interior_h != interior_w:
                        continue
                    
                    size = interior_h
                    if size not in fill_map:
                        continue
                    
                    fill_color = fill_map[size]
                    
                    # Fill interior black cells
                    for r in range(r1 + 1, r2):
                        for c in range(c1 + 1, c2):
                            if output[r, c] == 0:
                                output[r, c] = fill_color
                            
    return output
