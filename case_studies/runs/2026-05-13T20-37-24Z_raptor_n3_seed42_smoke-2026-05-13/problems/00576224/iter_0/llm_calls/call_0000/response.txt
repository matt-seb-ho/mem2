import numpy as np

def transform(input_grid: np.ndarray) -> np.ndarray:
    """
    Transforms a 2x2 input grid into a 6x6 output grid by tiling the input
    and its 90-degree clockwise rotation in alternating rows of 2x2 blocks.
    
    The output is constructed as a 3x3 grid of 2x2 blocks:
    - Rows 0-1 (block row 0): original input repeated horizontally 3 times.
    - Rows 2-3 (block row 1): input rotated 90° clockwise, repeated horizontally 3 times.
    - Rows 4-5 (block row 2): original input repeated horizontally 3 times.
    """
    # Ensure input is 2x2 (though the pattern is defined only for this size)
    assert input_grid.shape == (2, 2), "Input must be 2x2"
    
    # Create output grid of size 6x6, initialized with zeros (black)
    output = np.zeros((6, 6), dtype=input_grid.dtype)
    
    # Tile the original input horizontally 3 times
    original_tile = np.tile(input_grid, (1, 3))  # shape (2, 6)
    
    # Compute the 90-degree clockwise rotation
    rotated = np.rot90(input_grid, k=-1)  # clockwise rotation
    rotated_tile = np.tile(rotated, (1, 3))  # shape (2, 6)
    
    # Assign the three block rows
    output[0:2, :] = original_tile   # first block row
    output[2:4, :] = rotated_tile    # second block row
    output[4:6, :] = original_tile   # third block row
    
    return output
