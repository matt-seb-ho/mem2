import numpy as np

def transform(input_grid: np.ndarray) -> np.ndarray:
    """
    Transforms a 2x2 input grid into a 6x6 output grid by tiling and alternating
    the pattern in a checkerboard-like manner.

    The transformation rule observed from the examples:
    - The input is a 2x2 grid.
    - The output is a 6x6 grid.
    - The output is constructed by repeating the 2x2 input grid in a 3x3 tiling,
      but with a twist: the pattern alternates between the original 2x2 block
      and a version where the two rows are swapped (or equivalently, the block
      is flipped vertically) in a checkerboard pattern.

    More concretely, consider the output as a 3x3 arrangement of 2x2 blocks.
    Let the input grid be:
        a b
        c d
    Then the output blocks are:
        Block (0,0): a b    Block (0,1): a b    Block (0,2): a b
                     c d            c d            c d
        Block (1,0): c d    Block (1,1): c d    Block (1,2): c d
                     a b            a b            a b
        Block (2,0): a b    Block (2,1): a b    Block (2,2): a b
                     c d            c d            c d

    This matches both provided examples.
    """
    # Extract the four values from the 2x2 input
    a = input_grid[0, 0]
    b = input_grid[0, 1]
    c = input_grid[1, 0]
    d = input_grid[1, 1]

    # Create the two 2x2 blocks
    block1 = np.array([[a, b],
                       [c, d]])
    block2 = np.array([[c, d],
                       [a, b]])

    # Assemble the 6x6 output by stacking blocks in a 3x3 pattern
    # Row 0: block1, block1, block1
    row0 = np.hstack([block1, block1, block1])
    # Row 1: block2, block2, block2
    row1 = np.hstack([block2, block2, block2])
    # Row 2: block1, block1, block1
    row2 = np.hstack([block1, block1, block1])

    # Stack rows vertically
    output_grid = np.vstack([row0, row1, row2])

    return output_grid
