import numpy as np

def transform(input_grid: np.ndarray) -> np.ndarray:
    """
    Transforms a 2x2 input grid into a 6x6 output grid by tiling the original
    and its 90-degree clockwise rotation in alternating rows of 2x2 blocks.
    """
    # Ensure input is 2x2
    assert input_grid.shape == (2, 2), "Input grid must be 2x2"
    
    # Create the rotated version (clockwise 90 degrees)
    rotated = np.rot90(input_grid, k=-1)
    
    # Initialize output grid (6x6)
    output = np.zeros((6, 6), dtype=input_grid.dtype)
    
    # Tile the blocks: 3 rows and 3 columns of 2x2 blocks
    for i in range(3):          # block row index
        for j in range(3):      # block column index
            # Choose block: original for even rows, rotated for odd rows
            block = input_grid if i % 2 == 0 else rotated
            # Place the block at the appropriate position
            output[i*2:(i+1)*2, j*2:(j+1)*2] = block
    
    return output
