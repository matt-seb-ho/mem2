import numpy as np

def transform(input_grid: np.ndarray) -> np.ndarray:
    """
    Transforms a 2x2 input grid into a 6x6 output grid by repeating the input
    horizontally three times and vertically stacking three copies, alternating
    between the original and its horizontal mirror image for the middle block.
    """
    H, W = input_grid.shape
    out_H, out_W = 3 * H, 3 * W
    output = np.zeros((out_H, out_W), dtype=input_grid.dtype)

    for r in range(out_H):
        block_idx = r // H          # which vertical block (0, 1, or 2)
        row_in_block = r % H        # row index within that block
        if block_idx % 2 == 0:      # even blocks: use original row
            row_data = input_grid[row_in_block, :]
        else:                       # odd block: use horizontally flipped row
            row_data = input_grid[row_in_block, ::-1]
        # repeat the row three times horizontally
        output[r, :] = np.tile(row_data, 3)

    return output
