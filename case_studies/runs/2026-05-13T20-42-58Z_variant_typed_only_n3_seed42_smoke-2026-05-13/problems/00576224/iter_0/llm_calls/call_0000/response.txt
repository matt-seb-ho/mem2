import numpy as np

def transform(input_grid: np.ndarray) -> np.ndarray:
    """
    Transform a 2x2 input grid into a 6x6 output grid by tiling the input
    and its column-swapped version in a 3x3 block pattern:
    - Top and bottom rows of 2x2 blocks are the original input.
    - Middle row of 2x2 blocks is the input with columns swapped.
    Each 2x2 block is repeated horizontally three times to fill the width.
    """
    # Input is expected to be 2x2
    # Create the horizontally tiled version of the original (2 rows, 6 columns)
    original_tiled = np.tile(input_grid, (1, 3))
    
    # Create the column-swapped version (swap left and right columns)
    swapped = input_grid[:, ::-1]  # or np.fliplr(input_grid)
    swapped_tiled = np.tile(swapped, (1, 3))
    
    # Stack vertically: original, swapped, original
    output_grid = np.vstack([original_tiled, swapped_tiled, original_tiled])
    
    return output_grid
