import numpy as np

def transform(input_grid: np.ndarray) -> np.ndarray:
    """
    Given a 2x2 input grid, produce a 6x6 output grid by tiling the input
    horizontally three times and vertically three times, with the middle
    vertical block row being the horizontally flipped version of the input.
    """
    # Horizontal flip of the input (swap columns)
    flipped = np.fliplr(input_grid)
    
    # Top and bottom blocks: original input tiled horizontally 3 times
    top_bottom = np.tile(input_grid, (1, 3))   # shape (2, 6)
    
    # Middle block: flipped input tiled horizontally 3 times
    middle = np.tile(flipped, (1, 3))          # shape (2, 6)
    
    # Stack vertically: top, middle, top
    output_grid = np.vstack([top_bottom, middle, top_bottom])
    
    return output_grid
