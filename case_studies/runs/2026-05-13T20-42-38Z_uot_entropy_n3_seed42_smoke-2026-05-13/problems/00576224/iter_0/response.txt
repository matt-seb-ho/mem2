import numpy as np

def transform(input_grid: np.ndarray) -> np.ndarray:
    """
    Transforms a 2x2 input grid into a 6x6 output grid by tiling the input
    3 times in both dimensions and then horizontally flipping the middle
    two rows (the second block row).
    """
    H, W = input_grid.shape
    # Tile the input to create a 3H x 3W grid
    output = np.tile(input_grid, (3, 3))
    # Horizontally flip the rows corresponding to the middle block row
    output[H:2*H, :] = np.fliplr(output[H:2*H, :])
    return output
