import numpy as np

def transform(input_grid: np.ndarray) -> np.ndarray:
    """
    Transforms a 2x2 input grid into a 6x6 output grid.
    The output is formed by tiling the input and its horizontal mirror
    in alternating rows of 2x2 blocks: original, flipped, original.
    """
    # Ensure input is 2x2
    assert input_grid.shape == (2, 2), "Input grid must be 2x2"
    
    # Horizontal flip: reverse columns
    flipped = input_grid[:, ::-1]
    
    # Build output row by row
    rows = []
    # First block row (original)
    for r in range(2):
        row = np.tile(input_grid[r, :], 3)
        rows.append(row)
    # Second block row (flipped)
    for r in range(2):
        row = np.tile(flipped[r, :], 3)
        rows.append(row)
    # Third block row (original again)
    for r in range(2):
        row = np.tile(input_grid[r, :], 3)
        rows.append(row)
    
    return np.array(rows)
